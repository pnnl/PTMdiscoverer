 32def validate_study_folder(
 33    base_data_folder: str,
 34    *,
 35    base_prompt_filename: str = "base_prompt.txt",
 36    protein_metadata_filename: str = "protein_metadata.json",
 37    dataset_labels_filename: str = "dataset_labels.csv",
 38) -> None:
 39    """Validate that a study folder has all required inputs.
 40
 41    Raises FileNotFoundError with a descriptive message on the first
 42    missing item.
 43    """
 44    if not os.path.isdir(base_data_folder):
 45        raise FileNotFoundError(f"Study folder '{base_data_folder}' does not exist or is not a directory.")
 46    for required in (base_prompt_filename, protein_metadata_filename):
 47        path = os.path.join(base_data_folder, required)
 48        if not os.path.isfile(path):
 49            raise FileNotFoundError(f"Required file '{required}' not found in study folder '{base_data_folder}'.")
 50    labels_path = os.path.join(base_data_folder, dataset_labels_filename)
 51    if os.path.isfile(labels_path):
 52        msfragger = os.path.join(base_data_folder, "Results_MSFragger")
 53        if not os.path.isdir(msfragger):
 54            raise FileNotFoundError(
 55                f"'{dataset_labels_filename}' exists but 'Results_MSFragger/' directory "
 56                f"is missing in '{base_data_folder}'."
 57            )

 60def _write_prediction_files(records: list[dict[str, Any]], pred_base: str) -> list[dict[str, Any]]:
 61    """Filter records to valid positions and write JSON + CSV prediction files.
 62
 63    Returns the list of kept (valid) records. Does NOT mutate the input list.
 64    """
 65    import csv as _csv
 66
 67    kept: list[dict[str, Any]] = []
 68    invalid_count = 0
 69    minus1_count = 0
 70    for rec in records:
 71        raw = rec.get("PTM.Protein.Position", -1)
 72        try:
 73            val = int(raw)
 74        except Exception:  # noqa: BLE001
 75            val = -1
 76        if val > 0:
 77            kept.append(rec)
 78        else:
 79            invalid_count += 1
 80            if val == -1:
 81                minus1_count += 1
 82    if invalid_count:
 83        msg = f"Dropped {invalid_count} record(s) with invalid PTM.Protein.Position"
 84        if minus1_count:
 85            if minus1_count == invalid_count:
 86                msg += " (all were -1)"
 87            else:
 88                msg += f" ({minus1_count} were -1)"
 89        msg += f". Kept {len(kept)}."
 90        logger.warning(msg)
 91    if not kept:
 92        return []
 93
 94    json_path = pred_base + ".json"
 95    csv_path = pred_base + ".csv"
 96    with open(json_path, "w", encoding="utf-8") as jf:
 97        json.dump(kept, jf, indent=2)
 98    fieldnames = sorted({k for r in kept for k in r.keys()})
 99    with open(csv_path, "w", newline="", encoding="utf-8") as cf:
100        writer = _csv.DictWriter(cf, fieldnames=fieldnames)
101        writer.writeheader()
102        writer.writerows(kept)
103    return kept

139def process_protein(
140    ctx: InferenceContext,
141    protein: dict[str, Any],
142) -> tuple[bool, list[dict[str, Any]]]:
143    """Process one protein: optionally extract delta masses, invoke LLM, persist artifacts."""
144    uniprot_id: str | None = protein.get("uniprot_id")
145    name: str = str(protein.get("name", uniprot_id) or "")
146    delta_mass_file: str | None = protein.get("delta_mass_file") or None
147
148    # Resolve relative paths (stored as relative to base_data_folder for portability)
149    if delta_mass_file and not os.path.isabs(delta_mass_file):
150        delta_mass_file = os.path.join(ctx.base_data_folder, delta_mass_file)
151
152    metadata_changed_local = False
153
154    needs_extraction = uniprot_id and (not delta_mass_file or not os.path.exists(delta_mass_file))
155    if needs_extraction:
156        assert uniprot_id is not None  # narrowed by `needs_extraction` guard above
157        if ctx.msrun_table.empty:
158            logger.warning("Cannot extract delta masses for %s; dataset labels file missing. Skipping.", uniprot_id)
159            return False, []
160        extracted_df = extract_delta_masses(uniprot_id, ctx.msrun_table, ctx.base_data_folder)
161        if extracted_df.empty:
162            logger.warning("No qualifying delta masses extracted for %s; skipping.", uniprot_id)
163            return False, []
164        delta_mass_file = f"{ctx.base_data_folder}/{uniprot_id}_delta-masses.csv"
165        extracted_df.to_csv(delta_mass_file, index=False)
166        # Store as relative path for portability
167        protein["delta_mass_file"] = os.path.relpath(delta_mass_file, ctx.base_data_folder)
168        metadata_changed_local = True
169        logger.info("Extracted %d delta mass rows for %s -> %s", len(extracted_df), uniprot_id, delta_mass_file)
170
171    if not (uniprot_id and delta_mass_file and os.path.exists(delta_mass_file)):
172        logger.warning("Skipping %s; delta mass file unavailable after extraction attempt.", uniprot_id)
173        return metadata_changed_local, []
174
175    try:
176        delta_mass_df = pd.read_csv(delta_mass_file)
177    except Exception as exc:  # noqa: BLE001
178        logger.error("Failed reading %s: %s", delta_mass_file, exc)
179        return metadata_changed_local, []
180
181    prompt = build_prompt(ctx.base_prompt, name, delta_mass_df)
182    response = ctx.client.chat.completions.create(model=ctx.model, messages=[{"role": "user", "content": prompt}])
183    answer = response.choices[0].message.content or ""
184
185    # Log prompt + raw answer
186    ctx.log_handle.write(f"----- {uniprot_id} PROMPT -----\n{prompt}\n----- {uniprot_id} RESPONSE -----\n{answer}\n\n")
187
188    # Parse JSON block
189    records = parse_first_json_block(answer)
190    # Attempt to augment missing / invalid positions using extraction table data
191    if records:
192        try:
193            augment_protein_positions(records, delta_mass_df)
194        except Exception as exc:  # pragma: no cover
195            logger.warning("Failed deriving PTM.Protein.Position for %s: %s", uniprot_id, exc)
196        records = validate_ptm_records(records)
197    os.makedirs(ctx.run_dir, exist_ok=True)
198    if records:
199        pred_base = os.path.join(ctx.run_dir, f"ptm_predictions_{uniprot_id}_{ctx.timestamp}")
200        records = _write_prediction_files(records, pred_base)
201        logger.info("Wrote %d PTM predictions for %s -> %s.json/.csv", len(records), uniprot_id, pred_base)
202    else:
203        logger.info("No parsable JSON block for %s; no prediction files written.", uniprot_id)
204
205    _write_html(prompt, answer, os.path.join(ctx.run_dir, f"output_{uniprot_id}_{ctx.timestamp}.html"))
206
207    return metadata_changed_local, records

210def format_timestamp(dt: datetime | None = None) -> str:
211    """Return a unique timestamp string YYYY-MM-DD-HH-MM-SS-mmm.
212
213    Uses a module-level monotonic counter to guarantee uniqueness even
214    when called in a tight loop within the same millisecond.
215    """
216    global _ts_counter, _ts_last  # noqa: PLW0603
217    dt = dt or datetime.now()
218    ts = dt.strftime("%Y-%m-%d-%H-%M-%S-") + f"{int(dt.microsecond / 1000):03d}"
219    if ts == _ts_last:
220        _ts_counter += 1
221        ts = f"{ts}-{_ts_counter}"
222    else:
223        _ts_last = ts
224        _ts_counter = 0
225    return ts

228def run_inference(
229    base_data_folder: str = ".",
230    *,
231    model: str = "gpt-4o",
232    client: Any | None = None,
233    timestamp: str | None = None,
234    dataset_labels_filename: str = "dataset_labels.csv",
235    base_prompt_filename: str = "base_prompt.txt",
236    protein_metadata_filename: str = "protein_metadata.json",
237    base_url: str | None = None,
238) -> dict[str, Any]:
239    """Run inference over all proteins.
240
241    Assumptions:
242        All required input files (`dataset_labels_filename`, `base_prompt_filename`,
243        `protein_metadata_filename`) reside directly within `base_data_folder`.
244
245    Parameters:
246        base_data_folder: Root folder containing inputs and where outputs will be created.
247        model: LLM model name.
248        client: Optional pre-configured OpenAI-compatible client (constructed if None).
249        timestamp: Optional fixed timestamp (for tests); generated if None.
250        dataset_labels_filename: Filename (not path) of CSV listing MS run IDs.
251        base_prompt_filename: Filename of base prompt text file.
252        protein_metadata_filename: Filename of protein metadata JSON file.
253        base_url: Base URL for the OpenAI-compatible API endpoint.
254    """
255    validate_study_folder(
256        base_data_folder,
257        base_prompt_filename=base_prompt_filename,
258        protein_metadata_filename=protein_metadata_filename,
259        dataset_labels_filename=dataset_labels_filename,
260    )
261
262    if client is None:
263        api_key = os.environ.get("OPENAI_API_KEY")
264        if not api_key:
265            raise SystemExit("OPENAI_API_KEY environment variable is required.")
266        effective_base_url = base_url or os.environ.get(
267            "OPENAI_BASE_URL", "https://api.openai.com/v1"
268        )
269        client = openai.OpenAI(api_key=api_key, base_url=effective_base_url)
270
271    base_prompt_file = os.path.join(base_data_folder, base_prompt_filename)
272    protein_metadata_file = os.path.join(base_data_folder, protein_metadata_filename)
273    dataset_labels_file = os.path.join(base_data_folder, dataset_labels_filename)
274
275    with open(base_prompt_file, encoding="utf-8") as handle:
276        base_prompt = handle.read()
277
278    proteins = load_protein_metadata(protein_metadata_file)
279    try:
280        msrun_table = pd.read_csv(dataset_labels_file)
281    except FileNotFoundError:
282        msrun_table = pd.DataFrame()
283    if timestamp is None:
284        timestamp = format_timestamp()
285    run_dir = os.path.join(base_data_folder, "output", f"output_{timestamp}")
286    os.makedirs(run_dir, exist_ok=True)
287    log_path = os.path.join(run_dir, f"output_log_{timestamp}.txt")
288
289    metadata_changed_global = False
290    total_records = 0
291    with open(log_path, "w", encoding="utf-8") as log_handle:
292        ctx = InferenceContext(
293            client=client,
294            base_prompt=base_prompt,
295            model=model,
296            timestamp=timestamp,
297            base_data_folder=base_data_folder,
298            run_dir=run_dir,
299            log_handle=log_handle,
300            msrun_table=msrun_table,
301        )
302        for protein in proteins:
303            changed, records = process_protein(ctx, protein)
304            metadata_changed_global = metadata_changed_global or changed
305            total_records += len(records)
306
307    # Persist updated metadata if any changes occurred
308    if metadata_changed_global:
309        backup_path = protein_metadata_file + ".bak"
310        try:
311            if os.path.exists(protein_metadata_file):
312                os.replace(protein_metadata_file, backup_path)
313        except Exception as exc:  # pragma: no cover
314            logger.warning("Could not create metadata backup: %s", exc)
315        try:
316            with open(protein_metadata_file, "w", encoding="utf-8") as md_handle:
317                json.dump({"version": 1, "proteins": proteins}, md_handle, indent=4)
318            logger.info("Updated protein metadata written to %s", protein_metadata_file)
319        except Exception as exc:  # pragma: no cover
320            logger.error("Error writing updated metadata: %s", exc)
321
322    logger.info(
323        "DONE! Processed %d proteins; total parsed PTM records: %d. Artifacts in run directory: %s",
324        len(proteins),
325        total_records,
326        run_dir,
327    )
328
329    return {
330        "run_dir": run_dir,
331        "timestamp": timestamp,
332        "proteins_processed": len(proteins),
333        "total_ptm_records": total_records,
334        "metadata_updated": metadata_changed_global,
335    }

338def run_and_aggregate_multiple_inferences(
339    n: int = 10,
340    *,
341    base_data_folder: str = ".",
342    model: str = "gpt-4o",
343    base_url: str | None = None,
344    dataset_labels_filename: str = "dataset_labels.csv",
345    base_prompt_filename: str = "base_prompt.txt",
346    protein_metadata_filename: str = "protein_metadata.json",
347    client: Any | None = None,
348    aggregated_csv_path: str | None = None,
349    add_source_path: bool = True,
350    add_run_index: bool = True,
351) -> tuple[list[str], pd.DataFrame]:
352    """Run inference `n` times and aggregate all prediction CSVs (mandatory).
353
354    Returns:
355        (run_dirs, aggregated_dataframe)
356
357    Aggregated DataFrame provenance columns:
358        * `Source.File`  - absolute path to originating per-protein CSV (if `add_source_path`)
359        * `Run.Index`    - integer index corresponding to execution order (if `add_run_index`)
360        * `Protein.ID`   - parsed from per-protein prediction filename
361    """
362    run_dirs: list[str] = []
363    # Use a root timestamp for the multi-run batch (distinct from per-run timestamps)
364    batch_timestamp = format_timestamp()
365    for _ in range(n):
366        ts = format_timestamp()
367        run_inference(
368            base_data_folder=base_data_folder,
369            model=model,
370            client=client,
371            timestamp=ts,
372            dataset_labels_filename=dataset_labels_filename,
373            base_prompt_filename=base_prompt_filename,
374            protein_metadata_filename=protein_metadata_filename,
375            base_url=base_url,
376        )
377        run_dirs.append(os.path.join(base_data_folder, "output", f"output_{ts}"))
378
379    frames: list[pd.DataFrame] = []
380    for idx, rd in enumerate(run_dirs):
381        pattern = os.path.join(rd, "ptm_predictions_*.csv")
382        for csv_path in glob.glob(pattern):
383            try:
384                df = pd.read_csv(csv_path)
385            except Exception as exc:  # noqa: BLE001
386                logger.debug("Failed reading prediction CSV %s: %s", csv_path, exc)
387                continue
388            if add_source_path:
389                df["Source.File"] = os.path.abspath(csv_path)
390            if add_run_index:
391                df["Run.Index"] = idx
392            base = os.path.basename(csv_path)
393            if base.startswith("ptm_predictions_"):
394                no_prefix = base[len("ptm_predictions_") :]
395                stem = os.path.splitext(no_prefix)[0]
396                parts = stem.split("_")
397                if len(parts) >= 2:
398                    df["Protein.ID"] = "_".join(parts[:-1])
399            frames.append(df)
400    aggregated = pd.concat(frames, ignore_index=True) if frames else pd.DataFrame()
401    if aggregated_csv_path is None:
402        # Default path inside base_data_folder/output with batch timestamp
403        os.makedirs(os.path.join(base_data_folder, "output"), exist_ok=True)
404        aggregated_csv_path = os.path.join(
405            base_data_folder,
406            "output",
407            f"aggregated_predictions_{batch_timestamp}.csv",
408        )
409    try:
410        aggregated.to_csv(aggregated_csv_path, index=False)
411    except Exception as exc:  # pragma: no cover
412        logger.debug("Failed writing aggregated CSV %s: %s", aggregated_csv_path, exc)
413    return run_dirs, aggregated

214    Uses a module-level monotonic counter to guarantee uniqueness even
215    when called in a tight loop within the same millisecond.
216    """
217    global _ts_counter, _ts_last  # noqa: PLW0603
218    dt = dt or datetime.now()
219    ts = dt.strftime("%Y-%m-%d-%H-%M-%S-") + f"{int(dt.microsecond / 1000):03d}"
220    if ts == _ts_last:
221        _ts_counter += 1
222        ts = f"{ts}-{_ts_counter}"
223    else:
224        _ts_last = ts
225        _ts_counter = 0
226    return ts
