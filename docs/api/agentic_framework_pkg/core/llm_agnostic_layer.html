 74    def _parse_azure_root_endpoint(self, api_base: str) -> str:
 75        """Extracts the root Azure endpoint from a potentially deployment-specific URL."""

 93    async def _call_azure_sdk_acompletion(
 94        self,
 95        model_deployment_name: str,
 96        api_key: str,
 97        api_base: str,
 98        api_version: str,
 99        messages: list[dict[str, str]],
100        stream: bool,
101        **kwargs: Any,
102    ) -> Any:
103        azure_root_endpoint = self._parse_azure_root_endpoint(api_base)
104        logger.debug(
105            f"Calling Azure SDK acompletion. Deployment: {model_deployment_name}, Endpoint: {azure_root_endpoint}, Version: {api_version}"
106        )
107        try:
108            client = AsyncAzureOpenAI(
109                api_key=api_key,
110                azure_endpoint=azure_root_endpoint,
111                api_version=api_version,
112            )

114            response = await client.chat.completions.create(
115                model=model_deployment_name,
116                messages=cast(Any, messages),
117                stream=stream,
118                **sdk_kwargs,
119            )
120            await client.close()
121            return response
122        except (
123            OpenAI_APIError,
124            OpenAI_AuthenticationError,
125            OpenAI_RateLimitError,
126            OpenAI_BadRequestError,
127            OpenAI_NotFoundError,
128            OpenAI_APIConnectionError,
129            OpenAI_APITimeoutError,
130        ) as e:
131            logger.error(
132                f"Azure SDK acompletion error for deployment {model_deployment_name}: {e}",
133                exc_info=True,
134            )
135            raise LLMServiceError(f"Azure SDK error for deployment {model_deployment_name}: {e}") from e
136        except Exception as e:

138                f"Unexpected Azure SDK acompletion error for deployment {model_deployment_name}: {e}",

188    async def _call_litellm(self, method_name: str, model: str, llm_purpose: str, **kwargs: Any) -> Any:
189        """
190        Private helper to make calls to LiteLLM, centralizing credential and endpoint logic.
191        """
192        # Pop explicit credentials from kwargs first
193        # These will take precedence if provided by the caller directly.
194        explicit_api_key = kwargs.pop("api_key", None)
195        explicit_api_base = kwargs.pop("api_base", None)
196        explicit_api_version = kwargs.pop("api_version", None)  # For Azure
197
198        # Helper to get env vars: tries PURPOSE_SPECIFIC_VAR, then GENERIC_VAR
199        def get_env_config(base_var_name: str, purpose: str) -> str | None:
200            val = os.getenv(f"{purpose.upper()}_{base_var_name}")
201            if val:
202                return val
203            return os.getenv(base_var_name)
204
205        # Check for internal LLM provider FIRST (highest priority)
206        internal_llm_api_key = os.getenv("INTERNAL_LLM_API_KEY")
207        internal_llm_base_url = os.getenv("INTERNAL_LLM_BASE_URL")
208        internal_llm_model = os.getenv("INTERNAL_LLM_MODEL")
209        internal_llm_embedding_model = os.getenv("INTERNAL_LLM_EMBEDDING_MODEL")
210
211        # If internal LLM is configured, use it for all purposes
212        if internal_llm_api_key and internal_llm_base_url:
213            logger.debug(f"Using internal LLM provider for purpose '{llm_purpose}': {internal_llm_base_url}")
214            final_api_key = internal_llm_api_key
215            final_api_base = internal_llm_base_url
216            final_api_version = None
217            config_source_log = f"internal_llm (purpose: {llm_purpose})"
218
219            # Override model with internal LLM model if available
220            if llm_purpose == "embedding" and internal_llm_embedding_model:
221                model = internal_llm_embedding_model
222            elif internal_llm_model:
223                model = internal_llm_model
224
225            # LiteLLM requires provider prefix for custom API bases
226            # Since internal LLM is OpenAI-compatible, prefix with "openai/"
227            if not model.startswith("openai/"):
228                model = f"openai/{model}"
229
230            # Construct LiteLLM parameters for internal provider
231            litellm_params = {
232                "model": model,
233                "api_key": final_api_key,
234                "api_base": final_api_base,
235            }
236            litellm_params.update(kwargs)
237
238            log_params = {k: v for k, v in litellm_params.items() if k != "api_key"}
239            logger.debug(
240                f"Calling LiteLLM method: {method_name} for internal model '{model}' (purpose: {llm_purpose}). Params (key excluded): {log_params}"
241            )
242
243            # Call LiteLLM directly for internal provider
244            try:
245                if method_name == "acompletion":
246                    return await litellm.acompletion(**litellm_params)
247                elif method_name == "aembedding":
248                    return await litellm.aembedding(**litellm_params)
249                else:
250                    raise ValueError(f"Unsupported LiteLLM method: {method_name}")
251            except Exception as e:
252                logger.error(f"Internal LLM call failed: {e}", exc_info=True)
253                raise LLMServiceError(f"Internal LLM error: {e}") from e
254
255        # Fetch purpose-aware environment variables
256        azure_env_api_key = get_env_config("AZURE_API_KEY", llm_purpose)
257        azure_env_api_base = get_env_config(
258            "AZURE_API_BASE", llm_purpose
259        )  # Should be like https://YOUR_RESOURCE_NAME.openai.azure.com/
260        azure_env_api_version = get_env_config("AZURE_API_VERSION", llm_purpose)
261
262        openai_env_api_key = get_env_config("OPENAI_API_KEY", llm_purpose)
263        # openai_env_api_base = get_env_config("OPENAI_API_BASE", llm_purpose) # If needed for OpenAI proxies
264
265        ollama_env_api_base = get_env_config("OLLAMA_API_BASE", llm_purpose)
266
267        # Determine final credentials and endpoint
268        final_api_key = explicit_api_key
269        final_api_base = explicit_api_base
270        final_api_version = explicit_api_version
271
272        config_source_log = f"explicit_kwargs (purpose: {llm_purpose})"  # Base assumption
273
274        # Prioritize Azure if its environment variables are fully set and no explicit conflicting base is given
275        if azure_env_api_key and azure_env_api_base and azure_env_api_version:
276            logger.debug(
277                f"Using Azure environment variables for purpose '{llm_purpose}': api_key, api_base, api_version."
278            )
279            all_explicit_were_none_for_azure = (
280                explicit_api_key is None and explicit_api_base is None and explicit_api_version is None
281            )
282            if final_api_key is None:
283                final_api_key = azure_env_api_key
284            if final_api_base is None:
285                final_api_base = azure_env_api_base
286            if final_api_version is None:
287                final_api_version = azure_env_api_version
288            if all_explicit_were_none_for_azure:
289                config_source_log = f"azure_env (purpose: {llm_purpose})"
290
291        elif (
292            openai_env_api_key and final_api_key is None and final_api_base is None
293        ):  # Check final_api_base to ensure Azure or explicit base didn't set it
294            logger.debug(f"Using OpenAI environment variables for purpose '{llm_purpose}': api_key.")
295            # Fallback to OpenAI if Azure is not fully configured and no explicit base suggests another provider
296            final_api_key = openai_env_api_key
297            # if openai_env_api_base and final_api_base is None: final_api_base = openai_env_api_base # If supporting OPENAI_API_BASE
298            config_source_log = f"openai_env (purpose: {llm_purpose})"  # This implies no explicit and no Azure took precedence for key/base
299
300        # Handle Ollama if model indicates it and no other base is set
301        if model.startswith("ollama/") and final_api_base is None:
302            logger.debug(f"Using Ollama environment variables for purpose '{llm_purpose}': api_base.")
303            final_api_base = ollama_env_api_base or "http://localhost:11434"
304            # Update config_source_log only if it wasn't set by a higher priority source for the base
305            if (
306                config_source_log == f"explicit_kwargs (purpose: {llm_purpose})"
307                and explicit_api_base is None
308                and not (
309                    azure_env_api_key
310                    and azure_env_api_base
311                    and azure_env_api_version
312                    and explicit_api_key is None
313                    and explicit_api_base is None
314                    and explicit_api_version is None
315                )
316                and not (openai_env_api_key and explicit_api_key is None and explicit_api_base is None)
317            ):  # Ensure higher priority didn't set base
318                config_source_log = f"ollama_default (purpose: {llm_purpose})"
319
320        # Construct parameters for LiteLLM call
321        # The 'model' variable here is the one determined by the public calling method (e.g., final_model in acreate_embedding or agenerate_response)
322
323        is_azure_provider = config_source_log.startswith("azure_env") or (
324            final_api_base and "azure" in final_api_base.lower()
325        )
326
327        # Dispatch to Azure SDK if applicable, otherwise use LiteLLM
328        if is_azure_provider and not model.startswith("ollama/"):
329            logger.debug(
330                f"Using Azure SDK for purpose '{llm_purpose}' with model '{model}' (config source: {config_source_log})."
331            )
332            if not final_api_key or not final_api_base or not final_api_version:
333                raise LLMServiceError(
334                    f"Azure provider selected for purpose '{llm_purpose}' but missing one or more required credentials (API Key, Base URL, API Version)."
335                )
336
337            if method_name == "acompletion":
338                logger.debug(
339                    f"Calling Azure SDK acompletion for model '{model}' (purpose: {llm_purpose}). Config source: '{config_source_log}'."
340                )
341                # Pass all relevant kwargs directly to the SDK call
342                return await self._call_azure_sdk_acompletion(
343                    model_deployment_name=model,
344                    api_key=final_api_key,
345                    api_base=final_api_base,
346                    api_version=final_api_version,
347                    **kwargs,  # Pass messages, stream, and any other completion params via kwargs
348                )
349            elif method_name == "aembedding":
350                logger.debug(
351                    f"Calling Azure SDK aembedding for model '{model}' (purpose: {llm_purpose}). Config source: '{config_source_log}'."
352                )
353                # Pass all relevant kwargs directly to the SDK call
354                return await self._call_azure_sdk_aembedding(
355                    model_deployment_name=model,
356                    api_key=final_api_key,
357                    api_base=final_api_base,
358                    api_version=final_api_version,
359                    input_texts=kwargs.get("input", []),
360                    **kwargs,  # Note: acreate_embedding passes 'input_texts', _call_litellm receives it as 'input' via kwargs
361                )
362            else:
363                raise ValueError(f"Unsupported LiteLLM method: {method_name}")
364        else:
365            # Use LiteLLM for non-Azure providers or Ollama
366            logger.debug(
367                f"Using LiteLLM for purpose '{llm_purpose}' with model '{model}' (config source: {config_source_log})."
368            )
369            litellm_params = {
370                "model": model,
371                "api_key": final_api_key,
372                "api_base": final_api_base,
373            }
374            # Add api_version for Azure if LiteLLM is still handling an Azure case (e.g. if logic changes or for specific non-SDK handled Azure models)
375            # This part is now less likely to be hit for Azure main models.
376            if final_api_version and is_azure_provider:  # is_azure_provider check is a bit redundant here but safe
377                if not (final_api_base and "api-version=" in final_api_base.lower()):
378                    litellm_params["api_version"] = final_api_version
379                elif final_api_base and "api-version=" in final_api_base.lower():
380                    logger.debug(
381                        f"api-version found in api_base ('{final_api_base}'), not adding separate api_version from env var '{final_api_version}'."
382                    )
383
384            litellm_params.update(kwargs)  # Add all other passed-in kwargs
385
386            log_params = {k: v for k, v in litellm_params.items() if k != "api_key"}
387            logger.debug(
388                f"Calling LiteLLM method: {method_name} for model '{model}' (purpose: {llm_purpose}). Config source: '{config_source_log}'. Params (key excluded): {log_params}"
389            )
390
391            try:
392                if method_name == "acompletion":
393                    # Pass all relevant kwargs (including messages, stream) via litellm_params
394                    return await litellm.acompletion(**litellm_params)
395                elif method_name == "aembedding":
396                    # LiteLLM's aembedding response needs to be parsed
397                    embedding_response = await litellm.aembedding(**litellm_params)
398                    return [item.embedding for item in embedding_response.data]
399                else:
400                    raise ValueError(f"Unsupported LiteLLM method: {method_name}")
401            except litellm.exceptions.AuthenticationError as e:
402                logger.error(
403                    f"LiteLLM AuthenticationError for model {model} (purpose: {llm_purpose}, api_base: {final_api_base}, source: {config_source_log}): {e}",
404                    exc_info=True,
405                )
406                raise LLMServiceError(f"Authentication error for model {model} (purpose: {llm_purpose}): {e}") from e
407            except litellm.exceptions.APIError as e:  # Includes NotFoundError, RateLimitError etc.
408                logger.error(
409                    f"LiteLLM API error for model {model} (purpose: {llm_purpose}, api_base: {final_api_base}, source: {config_source_log}): {e}",
410                    exc_info=True,
411                )
412                raise LLMServiceError(f"LLM API error for model {model} (purpose: {llm_purpose}): {e}") from e
413            except Exception as e:
414                logger.error(
415                    f"Unexpected error during LiteLLM call for model {model} (purpose: {llm_purpose}, api_base: {final_api_base}, source: {config_source_log}): {e}",
416                    exc_info=True,
417                )
418                raise LLMServiceError(f"Unexpected error for model {model} (purpose: {llm_purpose}): {e}") from e

420    async def agenerate_response(
421        self,
422        messages: list[dict[str, str]],
423        llm_purpose: str,  # e.g., "streamlit", "rag"
424        model: str | None = None,
425        stream: bool = False,
426        **kwargs: Any,
427    ) -> Any | AsyncGenerator[Any, None]:
428        """
429        Generates a response from the specified LLM.
430
431        Args:
432            messages (list): A list of message objects, typically {"role": "user/assistant/system", "content": "..."}.
433            llm_purpose (str): The purpose of the LLM call (e.g., "streamlit", "rag"). Used for config and default model selection.
434            model (str, optional): The model identifier. If None, uses default for llm_purpose.
435            stream (bool): Whether to stream the response. Defaults to False.
436            **kwargs: Additional parameters to pass to the LiteLLM completion call (e.g., temperature, max_tokens).
437
438        Returns:
439            The response from the LLM. If streaming, this will be an async iterator.
440        Raises:
441            LLMServiceError: If an API error occurs.
442            ValueError: If no model is specified and no default model can be found.
443        """
444        final_model = model
445        if final_model is None:
446            final_model = self.default_models.get(llm_purpose) or self.default_models.get("generic")
447        if final_model is None:
448            raise ValueError(
449                f"No model specified and no default model found for purpose '{llm_purpose}' or generic default."
450            )
451
452        return await self._call_litellm(
453            "acompletion",
454            model=final_model,
455            llm_purpose=llm_purpose,
456            messages=messages,
457            stream=stream,
458            **kwargs,
459        )

461    async def acreate_embedding(
462        self, input_texts: list[str], model: str | None = None, **kwargs: Any
463    ) -> list[list[float]]:
464        """
465        Generates embeddings for a list of input texts using LiteLLM.
466        """
467        llm_purpose = "embedding"
468        final_model = model
469        if final_model is None:
470            final_model = self.default_models.get(llm_purpose) or self.default_models.get("generic")
471        if final_model is None:
472            raise ValueError(
473                f"No model specified and no default model found for purpose '{llm_purpose}' or generic default."
474            )
475
476        embedding_response = await self._call_litellm(
477            "aembedding",
478            model=final_model,
479            llm_purpose=llm_purpose,
480            input=input_texts,
481            **kwargs,
482        )
483
484        # Convert LiteLLM response to List[List[float]]
485        # Handle both dict and object response formats from LiteLLM
486        if isinstance(embedding_response, dict):
487            # Dict format: {"data": [{"embedding": [...], ...}], ...}
488            if "data" in embedding_response:
489                return [item["embedding"] for item in embedding_response["data"]]
490            else:
491                raise ValueError(f"Dict response missing 'data' key: {embedding_response}")
492        elif hasattr(embedding_response, "data"):
493            # Object format: EmbeddingResponse with .data attribute
494            data_items = embedding_response.data
495            # Check if data items are dicts or objects
496            if data_items and isinstance(data_items[0], dict):
497                return [item["embedding"] for item in data_items]
498            else:
499                return [item.embedding for item in data_items]
500        elif isinstance(embedding_response, list):
501            # Already converted (from Azure SDK path)
502            return embedding_response
503        else:
504            raise ValueError(f"Unexpected embedding response format: {type(embedding_response)}")

506    def _initialize_nvidia_client(self):
507        """Initializes the NVIDIA AsyncOpenAI client if API key is available."""
508        if self._nvidia_api_key:
509            try:
510                self._nvidia_async_client = openai.AsyncOpenAI(
511                    base_url=self._nvidia_base_url,
512                    api_key=self._nvidia_api_key,
513                )
514                self.is_nvidia_client_available = True
515                logger.info(
516                    f"NVIDIA AsyncOpenAI client initialized for vision model '{self._nvidia_default_vision_model}' at base URL '{self._nvidia_base_url}'."
517                )
518            except Exception as e:
519                logger.error(
520                    f"Failed to initialize NVIDIA AsyncOpenAI client: {e}",
521                    exc_info=True,
522                )
523                self._nvidia_async_client = None

486        # Handle both dict and object response formats from LiteLLM
487        if isinstance(embedding_response, dict):
488            # Dict format: {"data": [{"embedding": [...], ...}], ...}
489            if "data" in embedding_response:
490                return [item["embedding"] for item in embedding_response["data"]]
491            else:
492                raise ValueError(f"Dict response missing 'data' key: {embedding_response}")
493        elif hasattr(embedding_response, "data"):

495            data_items = embedding_response.data
496            # Check if data items are dicts or objects
497            if data_items and isinstance(data_items[0], dict):
498                return [item["embedding"] for item in data_items]
499            else:
500                return [item.embedding for item in data_items]
501        elif isinstance(embedding_response, list):
502            # Already converted (from Azure SDK path)
503            return embedding_response
504        else:
505            raise ValueError(f"Unexpected embedding response format: {type(embedding_response)}")
